% 译者注——单行内容
\newcommand{\TranslatorNote}[1]{
	\textcolor{lightorange}{\textbf{译者注：#1}}
}

% 章节主主题
\newcommand{\ChapterTitle}[1]{
	\clearpage
	\phantomsection
	\section*{#1}
	\addcontentsline{toc}{section}{#1}
	\vspace{-3ex}
	\noindent{\color{orangebold}\rule{\linewidth}{2pt}}\par
	\vspace{1ex}
}

% 章节子主题
\newcommand{\SubSectionTitle}[1]{
	\vspace{-2ex}
	\phantomsection
	\subsection*{#1}
	\vspace{-1ex}
	\addcontentsline{toc}{subsection}{#1}
}
